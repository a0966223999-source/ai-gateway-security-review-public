import test from 'node:test';
import assert from 'node:assert/strict';
import {planDag} from './dag-plan.mjs';
const node = (key, depends_on = []) => ({key, depends_on, prompt: `Compute ${key}`,
  acceptance_criteria: ['Check the saved result']});
const graph = nodes => ({version: 1, dag_id: 'test-original-1', goal: 'Test graph', nodes});
test('fork and join schedules C only after A and B', () => {
  const result = planDag(graph([node('C', ['A', 'B']), node('B'), node('A')]));
  assert.deepEqual(result.waves, [[['A', 'B']], [['C']]]);
  assert.equal(result.host_dag_enforced, false);
});
test('diamond and disconnected node', () => {
  assert.deepEqual(planDag(graph([node('A'), node('B', ['A']), node('C', ['A']),
    node('D', ['B', 'C']), node('E')])).waves, [[['A', 'E']], [['B', 'C']], [['D']]]);
});
test('original batch limit is three', () => {
  assert.deepEqual(planDag(graph(['E','D','C','B','A'].map(k => node(k)))).waves,
    [[['A','B','C'],['D','E']]]);
});
test('cycle refuses entire graph even with an independent root', () => {
  assert.throws(() => planDag(graph([node('A', ['B']),node('B', ['A']),node('C')])), /CYCLIC_DAG/);
});
test('missing dependency rejected', () => assert.throws(() => planDag(graph([node('A',['X'])])), /MISSING_DEPENDENCY/));
test('self dependency rejected', () => assert.throws(() => planDag(graph([node('A',['A'])])), /SELF_DEPENDENCY/));
test('duplicate node rejected', () => assert.throws(() => planDag(graph([node('A'),node('A')])), /DUPLICATE_NODE/));
test('duplicate edge rejected', () => assert.throws(() => planDag(graph([node('A'),node('B',['A','A'])])), /INVALID_NODE/));
test('invalid and oversized graphs rejected', () => {
  assert.throws(() => planDag(graph([])), /INVALID_DAG/);
  assert.throws(() => planDag(graph(Array.from({length:101},(_,i)=>node(`n${i}`)))), /INVALID_DAG/);
  assert.throws(() => planDag({...graph([node('A')]), dag_id:'../other'}), /INVALID_DAG/);
});
test('planning copies data and performs no dispatch', () => {
  const input=graph([node('A')]); const copy=JSON.stringify(input);
  const result=planDag(input);result.nodes[0].acceptance_criteria.push('extra');
  assert.equal(JSON.stringify(input),copy);assert.equal(result.automatic_dispatch,false);
});
